USAGE EXAMPLE
================================================================================

import { CrosswordGrid } from './components';

function PuzzleView() {
  const [cells, setCells] = useState(puzzleData.cells);
  const [selectedCell, setSelectedCell] = useState(null);
  const [direction, setDirection] = useState('across');
  const [activeClue, setActiveClue] = useState(null);

  return (
    <CrosswordGrid
      cells={cells}
      gridSize={8}
      cluesAcross={puzzleData.clues_across}
      cluesDown={puzzleData.clues_down}
      selectedCell={selectedCell}
      currentDirection={direction}
      activeClue={activeClue}
      onCellSelect={setSelectedCell}
      onCellChange={setCells}
      onDirectionChange={setDirection}
      onActiveClueChange={setActiveClue}
    />
  );
}
