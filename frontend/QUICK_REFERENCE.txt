COMMON PATTERNS
================================================================================

1. Generate a Puzzle
-------------------
import { apiClient } from '../services/api';
import { useAPI } from '../hooks/useAPI';

const { data, loading, error, execute } = useAPI<PuzzleGenerateResponse>();

const handleGenerate = async () => {
  await execute(() => 
    apiClient.generatePuzzle({
      topic: 'Science',
      grid_size: 8,
      difficulty: 'medium',
    })
  );
};

2. Load and Display Puzzle
--------------------------
import { usePuzzle } from '../hooks/usePuzzle';

const { puzzle, loadPuzzle, userCells, cellStates } = usePuzzle();

// Load puzzle
loadPuzzle(puzzleData);

// Access cells
const cell = getCellAt(userCells, row, col);

3. Handle Cell Input
-------------------
import { sanitizeLetter } from '../utils/validation';
import { usePuzzle } from '../hooks/usePuzzle';

const { setCellValue, selectCell } = usePuzzle();

const handleKeyDown = (e: React.KeyboardEvent) => {
  const letter = sanitizeLetter(e.key);
  if (letter) {
    setCellValue(row, col, letter);
  }
};

4. Navigate Grid
---------------
import { getNextCellInWord } from '../utils/keyboard';

const next = getNextCellInWord(
  currentRow,
  currentCol,
  'across',
  8,
  blockedCells
);

if (next) {
  selectCell(next.row, next.col);
}

5. Validate Solution
-------------------
import { apiClient } from '../services/api';

const result = await apiClient.validateSolution(puzzleId, {
  cells: userCells,
});

if (result.is_valid) {
  console.log('Puzzle solved!');
}

import { apiClient } from '../services/api';

================================================================================
TYPE REFERENCE
================================================================================

Cell
----
{
  row: number;
  col: number;
  value: string | null;
  is_blocked: boolean;
  number: number | null;
}

Clue
----
{
  number: number;
  direction: 'across' | 'down';
  text: string;
  answer: string | null;
  start_row: number;
  start_col: number;
  length: number;
}

Puzzle
------
{
  puzzle_id: string;
  topic: string;
  grid_size: number;
  cells: Cell[];
  clues_across: Clue[];
  clues_down: Clue[];
  word_count: number;
  fill_rate: number;
  difficulty: 'easy' | 'medium' | 'hard';
  created_at: string;
  metadata: Record<string, unknown>;
}

================================================================================
HOOK REFERENCE
================================================================================

useAPI<T>()
-----------
Returns: {
  data: T | null;
  loading: boolean;
  error: string | null;
  execute: (apiCall: () => Promise<T>) => Promise<T | null>;
  reset: () => void;
  clearError: () => void;
}

usePuzzle()
-----------
Returns: {
  // State
  puzzle: Puzzle | null;
  userCells: Cell[];
  selectedCell: { row: number; col: number } | null;
  currentDirection: Direction;
  activeClue: { number: number; direction: Direction } | null;
  cellStates: CellState[];
  gridState: GridState;
  isComplete: boolean;
  completionPercentage: number;
  
  // Actions
  loadPuzzle: (puzzle: Puzzle) => void;
  clearPuzzle: () => void;
  setCellValue: (row: number, col: number, value: string | null) => void;
  clearCell: (row: number, col: number) => void;
  resetGrid: () => void;
  selectCell: (row: number, col: number, direction?: Direction) => void;
  toggleDirection: () => void;
  selectClue: (clue: Clue) => void;
}

================================================================================
TROUBLESHOOTING
================================================================================

TypeScript errors:
  1. Run: npx tsc --noEmit
  2. Check import paths (use relative paths, not @types/)
  3. Verify all types are exported

Build errors:
  1. Run: npm run build
  2. Check for syntax errors
  3. Verify all dependencies are installed

API errors:
  1. Check backend is running on port 8000
  2. Verify VITE_API_BASE_URL in .env
  3. Check network tab in browser DevTools

Import errors:
  1. Use relative paths: '../types/puzzle'
  2. Don't use @types/ prefix
  3. Check tsconfig.json path aliases

================================================================================
BEST PRACTICES
================================================================================

1. Always validate user input with sanitizeLetter()
2. Check coordinates with isValidCoord() before accessing cells
3. Use getCellAt() instead of direct array access
4. Clone cells with cloneCells() before modifying
5. Wrap API calls in try-catch blocks
6. Use useAPI hook for consistent state management
7. Use type imports: import type { ... }
8. Keep cell state immutable
9. Validate form inputs before API calls
10. Handle loading and error states in UI
